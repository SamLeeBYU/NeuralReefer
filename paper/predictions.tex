% ----- local knobs (scoped to this file only) -----
\newcommand{\qualOGext}{.jpg}
\newcommand{\qualGText}{.png}
\newcommand{\qualPREext}{.png}

\newlength{\qualColW}   \setlength{\qualColW}{0.20\textwidth}
\newlength{\qualMetW}   \setlength{\qualMetW}{0.13\textwidth}
\newlength{\qualRowH}   \setlength{\qualRowH}{0.165\textheight}

% local gutters (scoped)
\setlength{\tabcolsep}{4pt}
\renewcommand{\arraystretch}{0.5}
\newcommand{\qualRowSpace}{0.1em}

% fixed-height, centered cell -- constrains BOTH height and width (with
% keepaspectratio picking whichever is tighter), so a gt/pred image whose
% aspect ratio isn't perfectly square (matplotlib's bbox_inches='tight' can
% crop asymmetrically per image) can never overflow the column width the
% way a height-only constraint would allow.
\newcommand{\qualCellImg}[2]{%
  \begin{minipage}[c][\qualRowH][c]{\qualColW}\centering
    \includegraphics[height=\dimexpr #2\qualRowH\relax, width=\qualColW, keepaspectratio]{#1}%
  \end{minipage}%
}

% column helpers
% og.jpg fills its full frame (no margin). gt/pred.png are saved on a 10x10in
% matplotlib canvas with axes at [0.05,0.05,0.9,0.9], so the visible coral
% content occupies 90% of the image width. Since \qualColW < \qualRowH the
% width constraint always binds with keepaspectratio, so we constrain og to
% 0.90\qualColW (matching the 90% visible content of gt/pred at \qualColW).
\newcommand{\qualOGcell}[1]{%
  \begin{minipage}[c][\qualRowH][c]{\qualColW}\centering
    \includegraphics[height=\qualRowH, width=0.90\qualColW, keepaspectratio]{fig9#1_og\qualOGext}%
  \end{minipage}}
\newcommand{\qualGTcell}[1]{%
  \qualCellImg{fig9#1_gt\qualGText}{1.00}}
\newcommand{\qualPREcell}[1]{%
  \qualCellImg{fig9#1_pred\qualPREext}{1.00}}

% metrics cell: #1=IoU #2=Dice #3=Prec #4=Rec
\newcommand{\qualStatCell}[4]{%
  \begin{minipage}[c][\qualRowH][c]{\qualMetW}\centering\small
    IoU:~#1\\[3pt]Dice:~#2\\[3pt]Prec:~#3\\[3pt]Rec:~#4
  \end{minipage}%
}

% one table row: #1 = letter (a/b/c/d), #2-#5 = IoU/Dice/Prec/Rec
\newcommand{\qualExampleRow}[5]{%
  \qualOGcell{#1} & \qualGTcell{#1} & \qualPREcell{#1} & \qualStatCell{#2}{#3}{#4}{#5} \\
  \addlinespace[\qualRowSpace]
}
